```python
@router.patch("/clusters/{cluster_id}/label", status_code=status.HTTP_200_OK)
async def set_cluster_label(cluster_id: UUID, body: ClusterLabelRequest) -> dict:
    """Set or clear the freeform nickname for a cluster. Empty string → null."""
    label = body.label.strip() if body.label else None
    if label and len(label) > 100:
        raise HTTPException(status_code=422, detail="label must be ≤ 100 characters")
    pool = await get_pool()
    async with pool.acquire() as conn:
        result = await conn.execute(
            "UPDATE unknown_clusters SET label = $1 WHERE id = $2",
            label,
            cluster_id,
        )
    if result == "UPDATE 0":
        raise HTTPException(status_code=404, detail="Cluster not found")
    return {"id": str(cluster_id), "label": label}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08B-01 | Tampering | `unknown_clusters.label` via PATCH | mitigate | Validate `len(label) <= 100` before write; strip whitespace; map empty→null at API layer |
| T-08B-02 | Spoofing | `PATCH /clusters/{id}/label` | mitigate | Endpoint inherits `require_token` from clustering router registration in main.py — no unauthenticated write possible |
| T-08B-03 | Injection | asyncpg parameterized query `$1, $2` | accept | asyncpg parameterized queries prevent SQL injection by design; `$1` is the label string, `$2` is the UUID — no string concatenation |
| T-08B-04 | Information Disclosure | digest.py caption | accept | Label is user-supplied content intended for Telegram; single-user system; no PII beyond what user explicitly typed |
